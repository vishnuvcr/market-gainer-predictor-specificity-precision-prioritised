//@version=6
strategy("NSE Breakout Surge ML Ensemble [v6]", overlay=true, initial_capital=100000, default_qty_type=strategy.percent_of_equity, default_qty_value=15, commission_type=strategy.commission.percent, commission_value=0.03, slippage=1, process_orders_on_close=true)

threshold_score = input.float(72.0, "ML Score Threshold (0-100)")
min_vol_surge = input.float(1.5, "Min Volume Surge Multiplier")
lookback_breakout = input.int(20, "Breakout Lookback")

target_1_pct = input.float(5.0, "Target 1 (%)")
target_2_pct = input.float(10.0, "Target 2 (%)")
target_3_pct = input.float(18.0, "Target 3 (%)")
atr_len = input.int(14, "ATR Length")
atr_sl_mult = input.float(1.5, "ATR SL Multiplier")
max_sl_cap_pct = input.float(4.0, "Max SL Cap (%)")

sma20 = ta.sma(close, 20)
sma50 = ta.sma(close, 50)
vol_sma20 = ta.sma(volume, 20)
vol_surge = vol_sma20 > 0 ? (volume / vol_sma20) : 1.0
rsi14 = ta.rsi(close, 14)
atr_val = ta.atr(atr_len)

highest_high = ta.highest(high, lookback_breakout)
float score = 0.0
if vol_surge >= min_vol_surge * 1.5
    score += 35.0
else if vol_surge >= min_vol_surge
    score += 25.0
if rsi14 >= 50.0 and rsi14 <= 78.0
    score += 25.0
if close > sma20 and sma20 > sma50
    score += 20.0
if close >= highest_high * 0.985
    score += 20.0

entry_condition = (score >= threshold_score) and (strategy.position_size == 0)
var float stop_price = na
var float tp1_price = na

if entry_condition
    float sl_dist = math.min(atr_val * atr_sl_mult, close * (max_sl_cap_pct / 100.0))
    stop_price := close - sl_dist
    tp1_price := close * (1.0 + target_1_pct / 100.0)
    strategy.entry("ML_Surge_Long", strategy.long)

if strategy.position_size > 0
    strategy.exit("TP1_Exit", from_entry="ML_Surge_Long", limit=tp1_price, stop=stop_price)

plot(sma20, "SMA 20", color=color.blue)
plot(sma50, "SMA 50", color=color.orange)
plotshape(entry_condition, "ML Surge", style=shape.triangleup, location=location.belowbar, color=color.green, size=size.normal)
